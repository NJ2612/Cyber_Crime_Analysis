Cybercrime Database ER Diagram

erDiagram
    CRIMES {
        int crime_id PK
        string type
        date date
        string city
        string state
        int severity
        string description
        string ip_address
    }
    SUSPECTS {
        int suspect_id PK
        string name
        int age
        string gender
        string address
        string criminal_history
    }
    VICTIMS {
        int victim_id PK
        string name
        int age
        string gender
        string address
    }
    EVIDENCE {
        int evidence_id PK
        int crime_id FK
        string type
        string description
        string file_path
    }
    FORECASTS {
        int id PK
        string city
        string state
        string month
        float risk_score
        string label
        datetime created_at
    }

    CRIMES ||--o{ EVIDENCE : has
    CRIMES }o--o{ SUSPECTS : involves
    CRIMES }o--o{ VICTIMS : affects
